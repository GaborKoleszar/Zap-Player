set(DOCTEST_DIR ${CMAKE_BINARY_DIR}/doctest)
if(NOT EXISTS ${DOCTEST_DIR}/doctest.h)
  file(DOWNLOAD https://raw.githubusercontent.com/doctest/doctest/v2.4.11/doctest/doctest.h
       ${DOCTEST_DIR}/doctest.h STATUS DOCTEST_STATUS)
  list(GET DOCTEST_STATUS 0 DOCTEST_CODE)
  if(NOT DOCTEST_CODE EQUAL 0)
    file(REMOVE ${DOCTEST_DIR}/doctest.h)
    message(FATAL_ERROR "Could not download doctest.h: ${DOCTEST_STATUS}")
  endif()
endif()
file(GLOB VP_TEST_SOURCES CONFIGURE_DEPENDS *.cpp)
add_executable(vp_tests ${VP_TEST_SOURCES})
target_include_directories(vp_tests PRIVATE ${DOCTEST_DIR})
target_link_libraries(vp_tests PRIVATE vp_core)
target_compile_definitions(vp_tests PRIVATE VP_MEDIA_DIR="${CMAKE_SOURCE_DIR}/bench/media")
vp_copy_mpv(vp_tests)
add_test(NAME vp_tests COMMAND vp_tests)
